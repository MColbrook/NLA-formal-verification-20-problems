module
public import Lean
@[expose] public section
open Lean
def positionJSON (p : Position) : Json := Json.mkObj [("line",toJson p.line),("column",toJson p.column)]
def rangeJSON (r : DeclarationRange) : Json := Json.mkObj [("start",positionJSON r.pos),("end",positionJSON r.endPos)]
unsafe def main : IO UInt32 := do
  initSearchPath (← findSysroot)
  let path ← findOLean `Challenge
  let (mainData, mainRegion) ← CompactedRegion.read (α := ModuleData) path #[]
  let (serverData, serverRegion) ← CompactedRegion.read (α := ModuleData)
    (OLeanLevel.server.adjustFileName path) #[mainRegion]
  let (privateData, privateRegion) ← CompactedRegion.read (α := ModuleData)
    (OLeanLevel.private.adjustFileName path) #[mainRegion, serverRegion]
  let owners ← IO.mkRef (mainRegion, serverRegion, privateRegion)
  -- The exported view may represent a theorem by its axiom-like signature;
  -- its original declaration kind is retained in the private module part.
  let constants := privateData.constants ++ serverData.constants ++ mainData.constants
  let rangeEntries : Array (Name × DeclarationRanges) := match serverData.entries.find? (fun item => item.1 == declRangeExt.name) with
    | some (_, entries) => unsafeCast entries
    | none => #[]
  let mut rows : Array Json := #[]
  for name in (#[`NLA.MF05.Paper.paper_real_bounded_positive_radius_growth, `NLA.MF05.Paper.paper_real_bounded_zero_radius_nilpotent, `NLA.MF05.Paper.paper_real_uniform_holder, `NLA.MF12.Paper.paper_main_realization, `NLA.MF12.Paper.paper_rational_realization, `NLA.RA02.Paper.paper_sup_worst_case_constants, `NLA.RA08.Paper.paper_spectral_failure, `NLA.RA08.Paper.verified_paper_nuclear_counterexample, `NLA.RA09.paper_ordered_matrix_norm_relative, `NLA.RA09.paper_ordered_sharpness, `NLA.RA09.paper_unordered_matrix_norm_relative, `NLA.RA09.paper_unordered_sharpness, `NLA.IE21.Paper.paper_random_rows, `NLA.IE22.Paper.paper_high_aspect_supremum_limit, `NLA.IE22.Paper.paper_universal_squared_rate, `Reference.LexTensorExecution.actual_lexicographic_paper_algorithm, `Reference.LexicographicFeasibilityBridge.product_eq_rowCount, `Reference.LexTensorCompletion.lexCard_eq_card, `NLA.TR27.Paper.paper_full_family, `NLA.TR27.Paper.coordinateProjection_surjective, `NLA.TR27.Paper.coordinateProjection_kernel, `NLA.TR27.Paper.imageHomogeneousMap_eq_zero_iff, `NLA.TR27.Paper.tensor_reps_nonzero, `NLA.IS03.paper_counterexample, `NLA.NM04.Paper.paper_rowland_wu_identity, `Reference.PF02UniformPositiveOrientation.paper_all_size_disconnected_witness, `Reference.MD06MainAssembly.Interface.actual_paper_md_main, `Reference.MD06FiniteProbability.actual_cubic_sample_nonempty, `IV04.Paper.Computational.actual_paper_iv_det, `Reference.IV04FullHullHardness.iv_hull_literal_np_hard, `NLA.MI19.Paper.paper_counterexample, `NLA.MI21.Paper.paper_counterexample, `NLA.MI22.Paper.paper_counterexample, `NLA.MI22.Paper.paper_conjecture_false, `NLA.MI22.singularValue._proof_1, `Complexity.NTM.trace._proof_6, `NLA.TR27.Paper.pivotIndex._proof_2, `NLA.TR27.Paper.recoveryIndex._proof_2, `NLA.TR27.Paper.recoveryOpenPoints._proof_2, `Reference.LexTensorCompletion.lexIndex._proof_5, `Reference.LexTensorCompletion.lexIndex._proof_2, `NLA.IE21.matrixMap._proof_6, `NLA.IE21.matrixMap._proof_4] : Array Name) do
    let info := constants.find? (fun info => info.name == name)
    let kind := match info with | some (.thmInfo _) => "theorem" | some _ => "other" | none => "missing"
    let owner := if info.isSome then some "Challenge" else none
    let ranges := (rangeEntries.find? (fun item => item.1 == name)).map (·.2)
    rows := rows.push <| Json.mkObj [("name",toJson name.toString),("kind",toJson kind),("module",toJson owner),("range",match ranges with | some r => rangeJSON r.range | none => Json.null),("selection",match ranges with | some r => rangeJSON r.selectionRange | none => Json.null)]
  let _ ← owners.get
  IO.println ("CHALLENGE_ANCHORS " ++ (toJson rows).compress)
  return 0
